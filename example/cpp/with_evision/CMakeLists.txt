cmake_minimum_required(VERSION 3.14)
project(EvisionOnnxExample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json 3.2.0 CONFIG REQUIRED)
set(ONNXRUNTIME_ROOT "$ENV{ONNXRUNTIME_ROOT}" CACHE PATH "ONNX Runtime native SDK")
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH)
find_library(ONNXRUNTIME_LIB onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH)
if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIB)
    message(FATAL_ERROR "Set ONNXRUNTIME_ROOT to the extracted native SDK")
endif()
if(MSVC AND MSVC_VERSION LESS 1920)
    if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../vision-runtime/include/ort_vs2017.cmake")
        include("${CMAKE_CURRENT_LIST_DIR}/../vision-runtime/include/ort_vs2017.cmake")
    else()
        include("${CMAKE_CURRENT_LIST_DIR}/../../../cpp/include/ort_vs2017.cmake")
    endif()
    dvs_ort_vs2017_headers("${ONNXRUNTIME_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/ort-v141-include")
    set(ONNXRUNTIME_INCLUDE_DIR "${CMAKE_CURRENT_BINARY_DIR}/ort-v141-include")
endif()
add_library(evision_classifier INTERFACE)
target_compile_features(evision_classifier INTERFACE cxx_std_17)
target_include_directories(evision_classifier INTERFACE "${CMAKE_CURRENT_LIST_DIR}" "${ONNXRUNTIME_INCLUDE_DIR}")
target_link_libraries(evision_classifier INTERFACE nlohmann_json::nlohmann_json "${ONNXRUNTIME_LIB}")
if(MSVC)
    if(MSVC_VERSION LESS 1916)
        message(FATAL_ERROR "Use Visual Studio 2017 15.9 (v141 14.16) or newer")
    endif()
    target_compile_options(evision_classifier INTERFACE /utf-8 /fp:precise /Zc:__cplusplus /Zc:noexceptTypes)
else()
    target_compile_options(evision_classifier INTERFACE -ffp-contract=off)
endif()
add_executable(evision_onnx_example main.cpp)
add_executable(evision_onnx_test self_test.cpp)
foreach(target evision_onnx_example evision_onnx_test)
    target_link_libraries(${target} PRIVATE evision_classifier)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_LIST_DIR}/../../assets" "$<TARGET_FILE_DIR:${target}>/assets")
    if(WIN32)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:${target}>/onnxruntime.dll")
    endif()
endforeach()
enable_testing()
add_test(NAME evision_bw8_example COMMAND evision_onnx_example)
add_test(NAME evision_bw8_contracts COMMAND evision_onnx_test)
set_tests_properties(evision_bw8_example evision_bw8_contracts PROPERTIES
    WORKING_DIRECTORY "${CMAKE_CURRENT_LIST_DIR}" TIMEOUT 60)

# Optional verification only; the two production targets above never link OpenCV.
option(EVISION_CHECK_OPENCV_PARITY "Compare BW8 preprocessing with OpenCV" OFF)
if(EVISION_CHECK_OPENCV_PARITY)
    find_package(OpenCV REQUIRED COMPONENTS core imgproc)
    add_executable(evision_preprocess_parity parity_test.cpp)
    target_link_libraries(evision_preprocess_parity PRIVATE evision_classifier ${OpenCV_LIBS})
    target_include_directories(evision_preprocess_parity PRIVATE ${OpenCV_INCLUDE_DIRS})
    if(WIN32)
        foreach(lib IN LISTS OpenCV_LIBS)
            if(TARGET ${lib})
                add_custom_command(TARGET evision_preprocess_parity POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "$<TARGET_FILE:${lib}>" "$<TARGET_FILE_DIR:evision_preprocess_parity>")
            endif()
        endforeach()
    endif()
    add_test(NAME evision_preprocessing_parity COMMAND evision_preprocess_parity)
endif()
