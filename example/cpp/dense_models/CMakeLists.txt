cmake_minimum_required(VERSION 3.14)
project(DenseModelExample LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(VISION_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(DEEPVISION_RUNTIME_SOURCE "" CACHE PATH "Path to the Vision C++ runtime source")
if(NOT DEEPVISION_RUNTIME_SOURCE)
    set(DEEPVISION_RUNTIME_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../cpp")
endif()
if(NOT EXISTS "${DEEPVISION_RUNTIME_SOURCE}/CMakeLists.txt")
    set(DEEPVISION_RUNTIME_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../vision-runtime")
endif()
if(NOT EXISTS "${DEEPVISION_RUNTIME_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Vision runtime source missing. Set DEEPVISION_RUNTIME_SOURCE.")
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
add_subdirectory(../nlohmann json_headers)
add_subdirectory("${DEEPVISION_RUNTIME_SOURCE}" vision_sdk EXCLUDE_FROM_ALL)
add_executable(dense_model_example main.cpp)
target_link_libraries(dense_model_example PRIVATE vision_inference)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT dense_model_example)
if(WIN32)
    get_filename_component(_ort_lib_dir "${ONNXRUNTIME_LIB}" DIRECTORY)
    unset(SEGMENT_ORT_DLL CACHE)
    find_file(SEGMENT_ORT_DLL NAMES onnxruntime.dll
        PATHS "${_ort_lib_dir}" "${_ort_lib_dir}/../bin" "${ONNXRUNTIME_ROOT}/lib" "${ONNXRUNTIME_ROOT}/bin" NO_DEFAULT_PATH)
    if(NOT SEGMENT_ORT_DLL)
        message(FATAL_ERROR "Matching onnxruntime.dll missing from ONNXRUNTIME_ROOT.")
    endif()
    foreach(target dense_model_example)
        if(MSVC)
            target_compile_options(${target} PRIVATE /utf-8)
        endif()
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SEGMENT_ORT_DLL}" $<TARGET_FILE_DIR:${target}>)
        foreach(opencv_target IN LISTS OpenCV_LIBS)
            if(TARGET ${opencv_target})
                get_target_property(_opencv_type ${opencv_target} TYPE)
                if(_opencv_type STREQUAL "SHARED_LIBRARY")
                    add_custom_command(TARGET ${target} POST_BUILD
                        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:${opencv_target}>" $<TARGET_FILE_DIR:${target}>)
                endif()
            endif()
        endforeach()
    endforeach()
endif()
